🤖 Multi-Agent System with Online MCP Servers

Connect real MCP servers + build a multi-agent orchestrator using LangGraph

┌──────────────────────────────────────────────────────────────────┐ │ MULTI-AGENT ORCHESTRATOR │ │ │ │ User Query → [Router Agent] → decides which specialist to call │ │ │ │ │ ┌────────────┼────────────────┐ │ │ ↓ ↓ ↓ │ │ [Research [Code [Document │ │ Agent] Agent] Agent] │ │ │ │ │ │ │ ↓ ↓ ↓ │ │ MCP: Web MCP: GitHub MCP: Filesystem │ │ Search Code Search Read/Write Files │ │ │ │ Each agent has access to specific MCP tools │ │ Router decides WHO handles the query │ │ Results are synthesized and returned │ └──────────────────────────────────────────────────────────────────┘ Online MCP Servers Used: • @modelcontextprotocol/server-brave-search — Web search • @modelcontextprotocol/server-github — GitHub repos • @modelcontextprotocol/server-filesystem — Local files

Step 2: Multi-Agent System (LangGraph)

┌─────────────┐ │ Router │ │ Agent │ └──────┬──────┘ │ (classifies intent) ┌────────────┼────────────────┐ ↓ ↓ ↓ ┌──────────────┐ ┌──────────────┐ ┌──────────────┐ │ Research │ │ Code │ │ Writer │ │ Agent │ │ Agent │ │ Agent │ │ │ │ │ │ │ │ Tools: │ │ Tools: │ │ Tools: │ │ • web_search│ │ • github_ │ │ • read_file │ │ • brave_api │ │ search │ │ • write_file │ └──────────────┘ └──────────────┘ └──────────────┘ │ │ │ └────────────┼────────────────┘ ↓ ┌─────────────┐ │ Synthesizer │ → Final Answer └─────────────┘ 
from typing import TypedDict, Literal, Annotated
from langgraph.graph import StateGraph, END
from langchain_groq import ChatGroq
import json

# ═══════════════════════════════════════════
# STATE
# ═══════════════════════════════════════════
class AgentState(TypedDict):
    query: str
    intent: str          # "research" | "code" | "write"
    research_result: str
    code_result: str
    write_result: str
    final_answer: str

# ═══════════════════════════════════════════
# LLM
# ═══════════════════════════════════════════
llm = ChatGroq(model="llama-3.3-70b-versatile", temperature=0.3)

# ═══════════════════════════════════════════
# ROUTER AGENT — classifies user intent
# ═══════════════════════════════════════════
def router_agent(state: AgentState) -> AgentState:
    prompt = f"""Classify this query into ONE category:
- "research" = needs web search, current info, news
- "code" = needs code examples, GitHub repos, technical implementation
- "write" = needs to create/edit documents or files

Query: {state['query']}
Return ONLY the category word."""
    
    response = llm.invoke(prompt)
    intent = response.content.strip().lower().replace('"', '')
    if intent not in ("research", "code", "write"):
        intent = "research"  # default
    return {"intent": intent}

# ═══════════════════════════════════════════
# SPECIALIST AGENTS
# ═══════════════════════════════════════════
def research_agent(state: AgentState) -> AgentState:
    """Uses web search MCP tool to find information."""
    # In real code: call brave-search MCP server
    # Simulated here for learning:
    prompt = f"""You are a research specialist. 
Search the web and provide comprehensive information about:
{state['query']}

Provide 3-5 key findings with sources."""
    response = llm.invoke(prompt)
    return {"research_result": response.content}

def code_agent(state: AgentState) -> AgentState:
    """Uses GitHub MCP tool to find code examples."""
    # In real code: call github MCP server to search repos
    prompt = f"""You are a code specialist.
Find relevant code implementations for:
{state['query']}

Provide working Python code examples."""
    response = llm.invoke(prompt)
    return {"code_result": response.content}

def write_agent(state: AgentState) -> AgentState:
    """Uses filesystem MCP tool to create documents."""
    # In real code: call filesystem MCP server to write files
    prompt = f"""You are a technical writer.
Create a well-structured document about:
{state['query']}"""
    response = llm.invoke(prompt)
    return {"write_result": response.content}

# ═══════════════════════════════════════════
# SYNTHESIZER — combines results
# ═══════════════════════════════════════════
def synthesizer(state: AgentState) -> AgentState:
    """Combine specialist output into final answer."""
    result = state.get("research_result") or state.get("code_result") or state.get("write_result") or ""
    prompt = f"""Synthesize this into a clear, concise answer:
{result}

Original question: {state['query']}"""
    response = llm.invoke(prompt)
    return {"final_answer": response.content}

# ═══════════════════════════════════════════
# ROUTING LOGIC
# ═══════════════════════════════════════════
def route_to_specialist(state: AgentState) -> Literal["research_agent", "code_agent", "write_agent"]:
    intent = state.get("intent", "research")
    if intent == "code": return "code_agent"
    if intent == "write": return "write_agent"
    return "research_agent"

# ═══════════════════════════════════════════
# BUILD GRAPH
# ═══════════════════════════════════════════
graph = StateGraph(AgentState)

graph.add_node("router", router_agent)
graph.add_node("research_agent", research_agent)
graph.add_node("code_agent", code_agent)
graph.add_node("write_agent", write_agent)
graph.add_node("synthesizer", synthesizer)

graph.set_entry_point("router")
graph.add_conditional_edges("router", route_to_specialist)
graph.add_edge("research_agent", "synthesizer")
graph.add_edge("code_agent", "synthesizer")
graph.add_edge("write_agent", "synthesizer")
graph.add_edge("synthesizer", END)

app = graph.compile()

# ═══════════════════════════════════════════
# RUN
# ═══════════════════════════════════════════
result = app.invoke({"query": "How to implement RAG with reranking in Python?"})
print(f"Intent: {result['intent']}")
print(f"Answer: {result['final_answer']}")

Step 3: Connecting REAL MCP Tools to Agents

# In production, each agent calls MCP tools like this:
# (using mcp client library)

from mcp.client import ClientSession
from mcp.client.stdio import stdio_client, StdioServerParameters

async def call_brave_search(query: str) -> str:
    """Call the Brave Search MCP server."""
    params = StdioServerParameters(
        command="npx",
        args=["-y", "@modelcontextprotocol/server-brave-search"],
        env={"BRAVE_API_KEY": "your-key"}
    )
    async with stdio_client(params) as (r, w):
        async with ClientSession(r, w) as session:
            await session.initialize()
            result = await session.call_tool("brave_web_search", {"query": query})
            return result.content[0].text

async def call_github_search(query: str) -> str:
    """Call the GitHub MCP server."""
    params = StdioServerParameters(
        command="npx",
        args=["-y", "@modelcontextprotocol/server-github"],
        env={"GITHUB_PERSONAL_ACCESS_TOKEN": "ghp_xxx"}
    )
    async with stdio_client(params) as (r, w):
        async with ClientSession(r, w) as session:
            await session.initialize()
            result = await session.call_tool("search_repositories", {"query": query})
            return result.content[0].text

Interview Answer

"How would you build a multi-agent system with MCP?"

"I'd use LangGraph as the orchestrator with a Router agent that classifies intent, then routes to specialist agents. Each specialist connects to specific MCP servers via stdio transport. For example: a Research agent connects to Brave Search MCP, a Code agent connects to GitHub MCP, and a Writer agent connects to Filesystem MCP. The router decides WHO handles the query, MCP handles HOW tools are discovered and called. Results flow back through a Synthesizer node that combines specialist outputs into a coherent response. This gives us modularity — add a new MCP server and the system gains new capabilities without changing agent logic."
